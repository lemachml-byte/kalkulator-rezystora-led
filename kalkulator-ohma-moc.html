// ====================================================== // KONTROLA SPÓJNOŚCI U – I – R // ====================================================== function sprawdzSpojnoscUIR(U, I, R) { // Sprawdzamy tylko wtedy, gdy wszystkie trzy wartości // zostały podane i są większe od zera. if ( !Number.isFinite(U) || !Number.isFinite(I) || !Number.isFinite(R) || U <= 0 || I <= 0 || R <= 0 ) { return null; } // Rezystancja wynikająca z prawa Ohma const R_obliczone = U / I; // Prąd wynikający z prawa Ohma const I_obliczone = U / R; // Tolerancja 1%. // Pozwala uniknąć zgłaszania błędu przez // niewielkie różnice zaokrągleń. const roznicaR = Math.abs(R_obliczone - R) / R; const roznicaI = Math.abs(I_obliczone - I) / I; // Jeżeli różnica przekracza 1%, // dane uznajemy za niespójne. if (roznicaR > 0.01 || roznicaI > 0.01) { return { zgodne: false, R_obliczone: R_obliczone, I_obliczone: I_obliczone }; } return { zgodne: true, R_obliczone: R_obliczone, I_obliczone: I_obliczone }; } // ====================================================== // WYŚWIETLENIE OSTRZEŻENIA // ====================================================== function pokazKontroleUIR(U, I, R) { const wynik = sprawdzSpojnoscUIR(U, I, R); // Usuwamy poprzednie ostrzeżenie, // aby nie pojawiały się kolejne kopie. const stareOstrzezenie = document.getElementById("ostrzezenie-ui-r"); if (stareOstrzezenie) { stareOstrzezenie.remove(); } if (!wynik || wynik.zgodne) { return; } const ostrzezenie = document.createElement("div"); ostrzezenie.id = "ostrzezenie-ui-r"; ostrzezenie.style.marginTop = "20px"; ostrzezenie.style.padding = "18px"; ostrzezenie.style.borderLeft = "5px solid #c62828"; ostrzezenie.style.borderRadius = "12px"; ostrzezenie.style.background = "#fff4f4"; ostrzezenie.style.color = "#7f1d1d"; ostrzezenie.innerHTML = `
UWAGA — DANE SĄ NIESPÓJNE
Dla podanych wartości:
U = ${U} V
I = ${I} A
R = ${R} Ω
Z prawa Ohma wynika:
R = U / I = ${U} / ${I} = ${wynik.R_obliczone.toFixed(2)} Ω
Natomiast dla wpisanego R = ${R} Ω prąd powinien wynosić:
I = U / R = ${U} / ${R} = ${wynik.I_obliczone.toFixed(4)} A
Sprawdź wartości U, I oraz R.
`; // Wstawiamy ostrzeżenie przed sekcją wzorów. const wzory = document.querySelector(".formulas"); if (wzory) { wzory.parentNode.insertBefore(ostrzezenie, wzory); } else { const wynikContainer = document.querySelector("main"); if (wynikContainer) { wynikContainer.appendChild(ostrzezenie); } } }
